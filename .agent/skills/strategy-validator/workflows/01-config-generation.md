### 1.2.2 Parallel Strategy

```yaml
# config/parallel_config.yaml
parallel:
  # Data parallel
  data_parallel: {{ dp }}

  # Tensor parallel
  tensor_parallel: {{ tp }}
  tensor_parallel_style: megatron  # or colossalai

  # Pipeline parallel
  pipeline_parallel: {{ pp }}
  pipeline_schedule: 1f1b  # or interleaved_1f1b
  num_microbatches: auto  # = global_batch_size / (dp * micro_batch_size)

  # Context parallel (if cp > 1)
  context_parallel: {{ cp }}
  context_parallel_algo: ulysses  # or ring_attention

  # Expert parallel (if ep > 1, MoE only)
  expert_parallel: {{ ep }}

  # FSDP
  fully_shard: {{ fsdp_level != None }}
  fsdp_level: {{ fsdp_level }}  # level1 / level2 / level3

  # Device mesh
  mesh_dims: [dp, pp, tp]  # Order matters: outermost → innermost

  # Activation checkpoint
  activation_checkpoint: {{ activation_checkpoint }}
  activation_checkpoint_method: uniform  # or block
  checkpoint_num_layers: 1  # checkpoint every N layers
```

## 1.3 Generate Launch Script

Create `run_validation_<timestamp>.sh`:

```bash
#!/bin/bash
# Generated by strategy-validator from PSA output
# Strategy: dp={{ dp }}, tp={{ tp }}, pp={{ pp }}, FSDP={{ fsdp_level }}
# Estimated: {{ throughput_tokens_per_sec }} tokens/s, {{ memory_per_device_gb }}GB

set -e

export WORLD_SIZE={{ device_count }}
export MASTER_ADDR=localhost
export MASTER_PORT=29500

# Ascend-specific (if NPU)
{% if device_type == "npu" %}
export HCCL_CONNECT_TIMEOUT=7200
export ASCEND_LAUNCH_BLOCKING=1
{% endif %}

# Generate torchrun command
torchrun \
  --nnodes=1 \
  --nproc_per_node={{ device_count }} \
  --master_addr=$MASTER_ADDR \
  --master_port=$MASTER_PORT \
  scripts/train_lm.py \
  --config config/generated_validation_config.yaml \
  --output_dir ./validation_output_$(date +%Y%m%d_%H%M%S) \
  2>&1 | tee validation_run.log
```

## 1.5 Pre-flight Validation

Before executing, check:

```python
def validate_config(config, psa_input):
    checks = []

    # Device count consistency
    dp * tp * pp * cp * ep == psa_input["hardware"]["device_count"]
    checks.append(("Device count match", ...))

    # Microbatch divisibility
    global_batch_size % (dp * micro_batch_size) == 0
    checks.append(("Microbatch divisible", ...))

    # Layer divisibility (PP)
    num_layers % pp == 0
    checks.append(("Layers divisible by PP", ...))

    # Head divisibility (TP)
    num_heads % tp == 0 and num_kv_heads % tp == 0
    checks.append(("Heads divisible by TP", ...))

    # Sequence divisibility (CP)
    seq_length % (cp * 2) == 0
    checks.append(("Seq divisible by CP", ...))

    # Model checkpoint exists
    os.path.exists(model_path) and os.path.isdir(model_path)
    checks.append(("Model checkpoint accessible", ...))

    return checks
```
